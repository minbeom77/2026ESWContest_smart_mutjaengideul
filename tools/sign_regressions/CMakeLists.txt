cmake_minimum_required(VERSION 3.20)
project(safehub_sign_regressions LANGUAGES CXX)
enable_testing()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(SIGN "${REPO}/SW/rpi4_sign_cpp")
set(VISION "${REPO}/SW/rpi4_vision_atlas")
if(MSVC)
  add_compile_options(/utf-8 /EHsc /UNDEBUG)
else()
  add_compile_options(-Wall -Wextra -UNDEBUG -ffp-contract=off)
endif()
find_package(Threads REQUIRED)
file(GLOB CORE_SOURCES "${SIGN}/src/*.cpp")
list(FILTER CORE_SOURCES EXCLUDE REGEX "/(main|jsonl_frame_hands|mediapipe_hand_detector)\\.cpp$")
add_library(sign_core STATIC ${CORE_SOURCES})
target_include_directories(sign_core PUBLIC "${SIGN}/include")
add_library(live_session STATIC "${VISION}/photo_pipeline/live_sign_session.cpp")
target_include_directories(live_session PUBLIC "${VISION}/photo_pipeline" "${SIGN}/include")
# Explicit opt-in diagnostic, not an accuracy test or part of default builds.
add_executable(camera_offline_probe EXCLUDE_FROM_ALL camera_offline_probe.cpp)
target_link_libraries(camera_offline_probe PRIVATE sign_core live_session Threads::Threads)
function(regression name source)
  add_executable(${name} "${source}")
  target_include_directories(${name} PRIVATE "${VISION}/photo_pipeline")
  target_link_libraries(${name} PRIVATE sign_core live_session Threads::Threads)
  add_test(NAME ${name} COMMAND ${name})
  set_tests_properties(${name} PROPERTIES WORKING_DIRECTORY "${REPO}" TIMEOUT 90)
endfunction()
foreach(name active_targets_static_support_test hand_coverage_quality_test sign_runtime_regression class_15_inference_parity_regression
    class_id_coverage_regression hand_input_regression temporal_resample_regression
    temp_eligibility_regression temp_min_frames_regression temp_positive_full_regression
    temp_negative_full_regression aircon_cold_last40_regression hot_rescue_regression)
  regression(${name} "${SIGN}/tests/${name}.cpp")
endforeach()
set(OPENCV_INCLUDE_ROOT "" CACHE PATH "Optional installed OpenCV include directory")
set(OPENCV_SOURCE_ROOT "" CACHE PATH "Optional OpenCV 4.9 source checkout")
set(OPENCV_BUILD_ROOT "" CACHE PATH "Optional built OpenCV 4.9 tree")
set(ORT_HEADERS "" CACHE PATH "Optional ONNX Runtime session headers")
set(PAHO_HEADERS "" CACHE PATH "Optional Paho MQTT C headers")
if((OPENCV_INCLUDE_ROOT OR (OPENCV_SOURCE_ROOT AND OPENCV_BUILD_ROOT)) AND ORT_HEADERS)
  if(OPENCV_INCLUDE_ROOT)
    set(CV_HEADERS "${OPENCV_INCLUDE_ROOT}")
  else()
    set(CV_HEADERS "${OPENCV_SOURCE_ROOT}/modules/core/include"
        "${OPENCV_SOURCE_ROOT}/modules/imgproc/include" "${OPENCV_BUILD_ROOT}")
  endif()
  # These are host tests with scripted model outputs, not ONNX recognition tests.
  regression(tracking_roi_runner_test "${VISION}/photo_pipeline/tracking_roi_runner_test.cpp")
  regression(vision_crop_nms_test "${VISION}/tests/vision_crop_nms_test.cpp")
  regression(recognition_light_test "${VISION}/tests/recognition_light_test.cpp")
  regression(landmark_preview_test "${VISION}/tests/landmark_preview_test.cpp")
  foreach(name tracking_roi_runner_test vision_crop_nms_test recognition_light_test landmark_preview_test)
    target_include_directories(${name} PRIVATE ${CV_HEADERS} "${ORT_HEADERS}")
    if(MSVC)
      target_link_libraries(${name} PRIVATE
        "${OPENCV_BUILD_ROOT}/lib/Release/opencv_core490.lib"
        "${OPENCV_BUILD_ROOT}/lib/Release/opencv_imgproc490.lib"
        "${OPENCV_BUILD_ROOT}/3rdparty/lib/Release/zlib.lib")
    else()
      target_link_libraries(${name} PRIVATE opencv_core opencv_imgproc)
    endif()
  endforeach()
  if(PAHO_HEADERS)
    add_library(vision_pipeline_syntax OBJECT "${VISION}/photo_pipeline/vision_stream_sign_decoupled.cpp")
    target_compile_definitions(vision_pipeline_syntax PRIVATE VISION_RAW_ONLY)
    target_include_directories(vision_pipeline_syntax PRIVATE ${CV_HEADERS} "${ORT_HEADERS}"
        "${PAHO_HEADERS}" "${SIGN}/include" "${VISION}/photo_pipeline")
  endif()
endif()
foreach(name live_sign_session_test live_sign_timing_test hand_observation_policy_test)
  regression(${name} "${VISION}/tests/${name}.cpp")
endforeach()
foreach(name classifier_frame_bridge_test palm_retry_policy_test tracking_roi_retry_policy_test async_translation_queue_test scenario_sign_policy_test)
  regression(${name} "${VISION}/photo_pipeline/${name}.cpp")
endforeach()
